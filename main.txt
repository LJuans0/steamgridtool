import tkinter as tk


def main():
    window = tk.Tk()
    window.title("SteamGrid Tool")
    window.geometry("700x500")

    title = tk.Label(
        window,
        text="SteamGrid Tool",
        font=("Segoe UI", 20)
    )
    title.pack(pady=30)

    window.mainloop()


if __name__ == "__main__":
    main()
